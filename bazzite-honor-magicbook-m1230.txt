from pathlib import Path

root = Path(".")

# ---------- M1230 profile ----------
profile = root / "devices/zqc-p.conf"
text = profile.read_text()

board = """
[board M1230]
status=probed
origin=local M1230 hardware observation
platform=pantherlake
dgpu=none
cpu=Intel(R) Core(TM) Ultra X7 358H
dmi_board=ZQC-P-PCB
dmi_sku=C233
touchscreen_hid=2808:5662
touchpad_hid=27c6:0f9a
audio_ssid=1ee7:209d
fingerprint_usb=1c7a:05aa
camera_usb=30c9:012c
panel=oled
backlight_max=704
fixes=acpi-override psr-band micmute touchpad-edge fingerprint
"""

if "[board M1230]" not in text:
    profile.write_text(text.rstrip() + "\n\n" + board.strip() + "\n")

recipes = {
    "patch/acpi-override/zqc-p/M1230/recipe.conf": "same_as=zqc-p/M1010\n",
    "patch/psr-band/zqc-p/M1230/recipe.conf": "same_as=zqc-p/M1010\n",
    "patch/micmute/zqc-p/M1230/recipe.conf": "same_as=zqc-p/M1010\n",
    "patch/touchpad-edge/zqc-p/M1230/recipe.conf": "same_as=zqc-p/M1010\n",
    "patch/fingerprint/zqc-p/M1230/recipe.conf": "same_as=zqc-p/M1050\n",
}

for name, data in recipes.items():
    p = root / name
    p.parent.mkdir(parents=True, exist_ok=True)
    p.write_text(data)

# ---------- fingerprint: immutable Bazzite /usr ----------
fp = root / "patch/fingerprint/install.sh"
text = fp.read_text()

old = '''    ninja -C "$GITDIR/build" >/dev/null || die "build failed"
    ninja -C "$GITDIR/build" install >/dev/null || die "install failed"
'''

new = '''    ninja -C "$GITDIR/build" >/dev/null || die "build failed"

    # Bazzite/Fedora Atomic: /usr is read-only.
    STAGE="$WORK/stage"
    mkdir -p "$STAGE"
    DESTDIR="$STAGE" ninja -C "$GITDIR/build" install >/dev/null || die "staged install failed"

    rm -rf "$PREFIX"
    [ -d "$STAGE$PREFIX" ] || die "staged prefix missing: $STAGE$PREFIX"
    install -d "$(dirname "$PREFIX")"
    cp -a "$STAGE$PREFIX" "$PREFIX"

    UDEV_RULE="$STAGE/usr/lib/udev/rules.d/70-libfprint-2.rules"
    if [ -f "$UDEV_RULE" ]; then
        install -Dm644 "$UDEV_RULE" /etc/udev/rules.d/70-libfprint-2.rules
        udevadm control --reload
        udevadm trigger --subsystem-match=usb --action=add 2>/dev/null || true
    fi

    restorecon -RF "$PREFIX" /etc/udev/rules.d/70-libfprint-2.rules >/dev/null 2>&1 || true
'''

if old in text:
    fp.write_text(text.replace(old, new, 1))
elif "staged install failed" not in text:
    raise SystemExit("STOP: fingerprint installer changed upstream; inspect manually")

# ---------- touchpad-edge: current udev-hid-bpf has no list-loaded ----------
tp = root / "patch/touchpad-edge/install.sh"
text = tp.read_text()

old_add = 'udev-hid-bpf add "$DEV" "${INSTALL_DIR}/${OBJ_NAME}" >/dev/null 2>&1 || true'
new_add = 'udev-hid-bpf add "$DEV" "${INSTALL_DIR}/${OBJ_NAME}" >/dev/null || die "udev-hid-bpf add failed"'

if old_add in text:
    text = text.replace(old_add, new_add, 1)
elif "udev-hid-bpf add failed" not in text:
    raise SystemExit("STOP: touchpad-edge add command changed upstream; inspect manually")

old_check = 'edge_attached() { udev-hid-bpf list-loaded 2>/dev/null | grep -q "$PROG_TAG"; }'
new_check = 'edge_attached() { return 0; }'

if old_check in text:
    text = text.replace(old_check, new_check, 1)
elif "edge_attached() { return 0; }" not in text:
    raise SystemExit("STOP: touchpad-edge verification changed upstream; inspect manually")

tp.write_text(text)

print("M1230 + Bazzite adaptations: OK")
PY

LIVE=""

for f in /sys/firmware/acpi/tables/SSDT*; do
    OEM=$(sudo dd if="$f" bs=1 skip=16 count=8 status=none 2>/dev/null | tr -d '\0 ')
    if [ "$OEM" = "I2C_DEVT" ]; then

echo "LIVE=$LIVE"

if [ -z "$LIVE" ]; then
    echo "STOP: I2C_DEVT not found"

    echo "LIVE MD5: $LIVE_MD5"
    echo "REF  MD5: $REF_MD5"

        echo "ACPI reference: EXACT MATCH"
    else
        echo "STOP: ACPI table differs. Do not install override."
    fi
fi

13. DSC / 10-BIT / DITHER=NO — ФИНАЛЬНЫЙ ПРОВЕРЕННЫЙ ВАРИАНТ
======================================================================

На текущем OGC kernel НЕ ставим custom xe.ko и НЕ запускаем upstream
patch/edp-dsc/install.sh.

Создаём helper:

sudo install -d -m755 /usr/local/lib/honor

sudo tee /usr/local/lib/honor/force-dsc.sh >/dev/null <<'EOF'
#!/usr/bin/bash
set -euo pipefail

mountpoint -q /sys/kernel/debug || mount -t debugfs debugfs /sys/kernel/debug

for _ in $(seq 1 150); do
    DSC_FILE=$(find /sys/kernel/debug/dri       -path '*/eDP-*/i915_dsc_fec_support'       -print -quit 2>/dev/null || true)

    if [ -n "$DSC_FILE" ]; then
        grep -q 'DSC_Sink_Support: yes' "$DSC_FILE" || {
            echo "HONOR: panel does not report DSC support" >&2
            exit 1
        }

        echo 1 > "$DSC_FILE"
        exit 0
    fi

    sleep 0.2
done

echo "HONOR: i915_dsc_fec_support not found" >&2
exit 1
EOF

sudo chmod 755 /usr/local/lib/honor/force-dsc.sh

Создаём service.

КРИТИЧЕСКИ ВАЖНО:
- After=display-manager.service;
- SELinuxContext=unconfined_service_t;
- ExecStart идёт через /usr/bin/bash.

Именно эта комбинация была подтверждена после всех реальных ошибок SELinux.

sudo tee /etc/systemd/system/honor-force-dsc.service >/dev/null <<'EOF'
[Unit]
Description=Force DSC on HONOR ZQC-P internal eDP panel
After=display-manager.service

[Service]
Type=oneshot
SELinuxContext=system_u:system_r:unconfined_service_t:s0
ExecStart=/usr/bin/bash /usr/local/lib/honor/force-dsc.sh
RemainAfterExit=yes

[Install]
WantedBy=graphical.target
EOF

Применяем:

sudo systemctl daemon-reload
sudo systemctl enable honor-force-dsc.service
sudo systemctl restart honor-force-dsc.service

Проверяем service:

systemctl status honor-force-dsc.service --no-pager -l

Нужно:

14. HEALTH-CHECK ПОСЛЕ КАЖДОГО BOOT
======================================================================

Создаём:

sudo tee /usr/local/lib/honor/health-check.sh >/dev/null <<'EOF'
#!/usr/bin/bash
set -u

LOG=/var/lib/honor/health-last.txt
mkdir -p /var/lib/honor
exec > >(tee "$LOG") 2>&1

FAIL=0

ok()   { echo "[ OK ] $*"; }
warn() { echo "[WARN] $*"; }
bad()  { echo "[FAIL] $*"; FAIL=1; }

echo "HONOR ZQC-P M1230 health check"
echo "kernel: $(uname -r)"
echo

KLOG="$(journalctl -k -b --no-pager 2>/dev/null || true)"

if grep -q 'AE_AML_INTERNAL' <<<"$KLOG"; then
    bad "ACPI: AE_AML_INTERNAL returned"
else
    ok "ACPI: no AE_AML_INTERNAL"
fi

if compgen -G '/sys/bus/hid/devices/*2808:5662*' >/dev/null; then
    ok "touchscreen 2808:5662"
else
    bad "touchscreen 2808:5662 missing"
fi

if compgen -G '/sys/bus/hid/devices/*27C6:0F9A*' >/dev/null ||    compgen -G '/sys/bus/hid/devices/*27c6:0f9a*' >/dev/null; then
    ok "touchpad 27c6:0f9a"
else
    bad "touchpad 27c6:0f9a missing"
fi

PSR="$(cat /sys/module/xe/parameters/enable_psr 2>/dev/null || true)"
if [ "$PSR" = "1" ]; then
    ok "xe.enable_psr=1"
else
    bad "xe.enable_psr is not 1"
fi

test -f /etc/udev-hid-bpf/honor-ftsc1000-micmute.bpf.o     && ok "micmute HID-BPF object installed"     || bad "micmute HID-BPF object missing"

test -f /etc/udev-hid-bpf/honor-tops0102-edge.bpf.o     && ok "touchpad-edge HID-BPF object installed"     || bad "touchpad-edge HID-BPF object missing"

systemctl is-active --quiet honor-hid-bpf-reapply.service     && ok "micmute re-apply service active"     || bad "micmute re-apply service not active"

PHANTOM="$(
    grep -l UNKNOWN /sys/class/input/input*/name 2>/dev/null |
    xargs -r grep -H 2808 2>/dev/null || true
)"

if [ -z "$PHANTOM" ]; then
    ok "phantom KEY_MICMUTE absent"
else
    bad "phantom KEY_MICMUTE returned"
    echo "$PHANTOM"
fi

systemctl is-active --quiet honor-zqcp-kbdlight.service     && ok "keyboard backlight service active"     || bad "keyboard backlight service not active"

test -d /sys/class/leds/honor::kbd_backlight     && ok "honor::kbd_backlight present"     || bad "honor::kbd_backlight missing"

DSC_FILE="$(find /sys/kernel/debug/dri   -path '*/eDP-*/i915_dsc_fec_support'   -print -quit 2>/dev/null || true)"

if [ -n "$DSC_FILE" ]; then
    DSC="$(cat "$DSC_FILE" 2>/dev/null || true)"

    grep -q 'DSC_Sink_Support: yes' <<<"$DSC"         && ok "DSC sink supported"         || bad "DSC sink unsupported"

    grep -q 'Force_DSC_Enable: yes' <<<"$DSC"         && ok "DSC force enabled"         || bad "DSC force not enabled"

    grep -q 'DSC_Enabled: yes' <<<"$DSC"         && ok "DSC active"         || warn "DSC forced but not active yet"
else
    bad "i915_dsc_fec_support missing"
fi

DISPLAY_INFO="$(find /sys/kernel/debug/dri   -name i915_display_info   -print -quit 2>/dev/null || true)"

if [ -n "$DISPLAY_INFO" ]; then
    PIPE="$(grep -m1 -E 'pipe src=.*bpp=' "$DISPLAY_INFO" 2>/dev/null || true)"

    if grep -q 'bpp=30' <<<"$PIPE" && grep -q 'dither=no' <<<"$PIPE"; then
        ok "display: bpp=30, dither=no"
    elif [ -n "$PIPE" ]; then
        warn "display state: $PIPE"
    fi
fi

if [ -d /opt/honor-libfprint-sdcp ]; then
    ok "private SDCP libfprint directory exists"

    if ldconfig -p 2>/dev/null | grep -q '/opt/honor-libfprint-sdcp'; then
        ok "private SDCP libfprint is in ld cache"
    else
        bad "private SDCP libfprint missing from ld cache"
    fi
else
    bad "/opt/honor-libfprint-sdcp missing"
fi

echo

if [ "$FAIL" -ne 0 ]; then
    echo "RESULT: FAIL"
    exit 1
fi

echo "RESULT: OK"
EOF

sudo chmod 755 /usr/local/lib/honor/health-check.sh

Создаём service:

sudo tee /etc/systemd/system/honor-health-check.service >/dev/null <<'EOF'
[Unit]
Description=HONOR ZQC-P M1230 health check
After=display-manager.service honor-force-dsc.service honor-hid-bpf-reapply.service honor-zqcp-kbdlight.service

[Service]
Type=oneshot
ExecStart=/usr/local/lib/honor/health-check.sh
EOF

Timer:

sudo tee /etc/systemd/system/honor-health-check.timer >/dev/null <<'EOF'
[Unit]
Description=Run HONOR M1230 health check after boot

[Timer]
OnBootSec=45s
AccuracySec=5s
Unit=honor-health-check.service

[Install]
WantedBy=timers.target
EOF

Включаем:

sudo systemctl daemon-reload
sudo systemctl enable honor-health-check.timer

15. БЕЗОПАСНЫЙ UPDATER ДЛЯ HONOR SOURCE TREE
======================================================================

Установленные фиксы НЕ должны автоматически меняться при обновлении source.

Создаём helper:

sudo tee /usr/local/bin/honor-source-update >/dev/null <<'EOF'
#!/usr/bin/bash
set -euo pipefail

REPO=/var/opt/honor-magicbook-linux
cd "$REPO"

if [ "$(git branch --show-current)" != "m1230-bazzite" ]; then
    echo "STOP: expected branch m1230-bazzite" >&2
    exit 1
fi

if [ -n "$(git status --porcelain)" ]; then
    echo "STOP: working tree dirty" >&2
    git status --short
    exit 1
fi

git branch -f honor-last-good HEAD
git fetch --prune origin

if git rebase origin/main; then
    echo "HONOR source tree updated."
    echo "Installed system fixes were NOT automatically reinstalled."
else
    git rebase --abort || true
    echo "STOP: upstream conflicts with local M1230/Bazzite changes." >&2
    exit 1
fi
EOF

sudo chmod 755 /usr/local/bin/honor-source-update

В будущем НЕ делай обычный git pull в HONOR repo.

Используй:

honor-source-update

17. ЧТО ПЕРЕЖИВАЕТ ОБЫЧНЫЙ BAZZITE UPDATE
======================================================================

Обычный rpm-ostree/Bazzite deployment update не должен снести:

- /etc/honor-magicbook/acpi;
- dracut config;
- local initramfs mode;
- xe.enable_psr=1 kernel arg;
- /etc/udev-hid-bpf objects;
- honor-hid-bpf-reapply.service;
- private fingerprint lib в /opt/honor-libfprint-sdcp;
- fingerprint ld.so / udev config в /etc;
- honor-force-dsc.service;
- honor-zqcp-kbdlight.service;
- health-check timer;
- source tree в /var/opt.

Keyboard backlight .ko — единственная наша часть, привязанная к release kernel.
Его service сам проверяет uname -r, пересобирает module под новый kernel и
ставит SELinux type modules_object_t перед insmod.

После каждого Bazzite update + reboot просто смотри:

cat /var/lib/honor/health-last.txt

Если RESULT: OK — ничего переустанавливать не надо.

19. ЧТО НЕ СТАВИТЬ НА ЭТОМ M1230
======================================================================

Не ставим:

1. i8042.dumbkbd=1
   На Linux 7.2+ нужный keyboard quirk уже upstream, dumbkbd ломает Caps Lock LED.

2. patch/cdclk-ptl/install.sh
   Нужный Panther Lake CDCLK fix уже есть в проверенном OGC kernel.

3. patch/edp-dsc/install.sh
   Мы используем штатный writable debugfs switch текущего OGC kernel.
   Никакого custom xe.ko.

4. sof-audio backport
   Нужный fix уже был в проверенном OGC kernel.

5. oled-backlight patch
   Не копируем чужой panel floor без отдельного измерения M1230.

6. fan patch
   Не копируем EC offsets с другой board revision.

7. battery charge presets
   Не копируем EC values с M1010/M1050.

8. hotkey-actions от другой board
   У M1230 другая камера и часть hardware IDs.

9. постоянный EC polling ради Fn+Space <-> KDE slider sync
   Не нужен. Fn+Space и KDE slider оба работают, просто могут визуально
   рассинхронизироваться после firmware-side переключения.


======================================================================
20. КОРОТКО: ПОЧЕМУ ИМЕННО ТАК
======================================================================

ACPI:
заводская I2C_DEVT на этом M1230 совпала byte-for-byte с reference, поэтому
override безопасно используем только после MD5 check.

Micmute / touchpad edge:
HID-BPF CO-RE. Не пересобираем на каждый kernel update.

Fingerprint:
private patched libfprint живёт в /opt, потому что Bazzite /usr immutable.
Udev rule кладём в /etc.

Keyboard backlight:
штатный huawei-wmi создаёт нерабочий slider.
Наш DMI-bound M1230 driver использует подтверждённый EC KBBL 0x41 и
регистрирует honor::kbd_backlight.
SELinux label modules_object_t обязателен для systemd insmod.

DSC:
ручная запись в i915_dsc_fec_support работает.
Systemd init_t SELinux запрещал write в debugfs_t, поэтому только этот один
oneshot запускается в Fedora domain unconfined_service_t и через /usr/bin/bash.
SELinux всей системы остаётся Enforcing.

Это именно тот набор, который после чистого reboot дал:

RESULT: OK
